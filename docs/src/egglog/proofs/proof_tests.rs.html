11    /// The proof encoder reads body variables' `term_proof`s from the RHS via
12    /// `:unsafe-seminaive` lookups. Assert this produces the same database as
13    /// the safe baseline (the same rules annotated `:naive`), for a hardcoded
14    /// handful of files (running it across all tests would be too slow).
15    #[test]
16    fn unsafe_seminaive_matches_naive() {
17        let files = [
18            "tests/calc.egg",
19            "tests/integer_math.egg",
20            "tests/fibonacci-demand.egg",
21            "tests/until.egg",
22        ];
23
24        for file in files {
25            let source = std::fs::read_to_string(file)
26                .unwrap_or_else(|e| panic!("couldn't read {file}: {e}"));
27
28            // Guard against a vacuous comparison: the two encodings must differ.
29            let encode = |naive: bool| -> String {
30                let mut egraph = crate::EGraph::new_with_proofs();
31                egraph.proof_state.force_proof_naive = naive;
32                egraph
33                    .resolve_program(Some(file.to_string()), &source)
34                    .unwrap_or_else(|e| panic!("{file} resolve (naive={naive}) failed: {e}"))
35                    .iter()
36                    .map(|cmd| cmd.to_string())
37                    .collect::<Vec<_>>()
38                    .join("\n")
39            };
40            assert!(
41                encode(false).contains(":unsafe-seminaive") && encode(false) != encode(true),
42                "expected {file} to exercise the `:unsafe-seminaive` encoding path"
43            );
44
45            // `print-size` summarizes the whole database (per-function row
46            // counts, sorted) deterministically.
47            let program = format!("{source}\n(print-size)");
48
49            let run = |naive: bool| -> Vec<CommandOutput> {
50                let mut egraph = crate::EGraph::new_with_proofs();
51                egraph.proof_state.force_proof_naive = naive;
52                egraph
53                    .parse_and_run_program(Some(file.to_string()), &program)
54                    .unwrap_or_else(|e| panic!("{file} (naive={naive}) failed: {e}"))
55            };
56
57            let unsafe_seminaive = CommandOutput::snapshot_stable_under_proof_encoding(&run(false));
58            let naive = CommandOutput::snapshot_stable_under_proof_encoding(&run(true));
59
60            assert_eq!(
61                unsafe_seminaive, naive,
62                ":unsafe-seminaive and :naive proof encodings disagree for {file}"
63            );
64        }
65    }

67    /// A user rule marked `:naive` must stay `:naive` through proof encoding;
68    /// dropping it would silently switch the rule to seminaive evaluation.
69    #[test]
70    fn proof_encoding_preserves_naive() {
71        // The second case binds an eq-sort body var, whose `term_proof` RHS
72        // read would otherwise force `:unsafe-seminaive`. Both must stay naive.
73        let cases = [
74            r#"(relation r (i64))
75               (relation s (i64))
76               (rule ((r x)) ((s x)) :naive :name "keep")"#,
77            r#"(sort Math)
78               (constructor Num (i64) Math)
79               (constructor Neg (Math) Math)
80               (relation seen (Math))
81               (rule ((Neg m)) ((seen m)) :naive :name "keep")"#,
82        ];
83        for source in cases {
84            let mut egraph = crate::EGraph::new_with_proofs();
85            let resolved = egraph.resolve_program(None, source).unwrap();
86            let rule = resolved
87                .iter()
88                .find_map(|c| match c {
89                    ResolvedCommand::Rule { rule } if rule.name == "keep" => Some(rule),
90                    _ => None,
91                })
92                .expect("instrumented rule not found");
93            assert_eq!(
94                rule.eval_mode,
95                RuleEvalMode::Naive,
96                "proof encoding did not preserve :naive for:\n{source}"
97            );
98        }
99    }
